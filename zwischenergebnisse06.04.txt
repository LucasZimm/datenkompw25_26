Code Ex-Golomb:


void EntropyEncoder::ex_golomb_rice_encode(unsigned n, unsigned gtNIdx, unsigned binIdx)
{
    int p = (int)std::floor(std::log2((double)(n + 1)));

    for (int i = 0; i < p; i++)
        aenc.encBin(m_pmfExpCtx[gtNIdx][std::min<unsigned>(N-1, binIdx++)], 1);
    aenc.encBin(m_pmfExpCtx[gtNIdx][std::min<unsigned>(N-1, binIdx++)], 0);

    if (p > 0) {
        unsigned suffix = n - ((1 << p) - 1);
        aenc.encBits(suffix, p);
    }
}

void EntropyDecoder::ex_golomb_rice_decode(unsigned& n, unsigned gtNIdx, unsigned binIdx)
{
    int p = 0;
    while (adec.decBin(m_pmfExpCtx[gtNIdx][std::min<unsigned>(N-1, binIdx++)])) {
        p++;
    }

    if (p > 0) {
        unsigned suffix = adec.decBits(p);
        n = (1 << p) - 1 + suffix;
    } else {
        n = 0;
    }
}
